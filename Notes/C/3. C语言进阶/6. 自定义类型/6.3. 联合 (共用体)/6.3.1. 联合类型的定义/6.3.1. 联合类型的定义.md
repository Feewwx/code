#####
- 联合的特征是所有成员共用一块空间,所以联合也叫共用体

```c
#include <stdio.h>

union Un {
    int a;
    char c;
};

int main() {
    union Un u;
    printf("%d\n", sizeof(u));  // 4

    printf("%p\n", &u);    // 0x7fff02436554
    printf("%p\n", &u.a);  // 0x7fff02436554
    printf("%p\n", &u.c);  // 0x7fff02436554

    return 0;
}
```
